cmake_minimum_required(VERSION 3.20)
project(robot_wamr C ASM)
if(NOT WAMR_ROOT_DIR OR NOT FW_ROOT)
  message(FATAL_ERROR "WAMR_ROOT_DIR and FW_ROOT are required")
endif()
set(WAMR_BUILD_PLATFORM "robot-stm32")
set(WAMR_BUILD_TARGET "THUMB")
set(WAMR_BUILD_INTERP 1)
set(WAMR_BUILD_FAST_INTERP 0)
set(WAMR_BUILD_AOT 0)
set(WAMR_BUILD_JIT 0)
set(WAMR_BUILD_LIBC_BUILTIN 0)
set(WAMR_BUILD_LIBC_WASI 0)
set(WAMR_BUILD_THREAD_MGR 0)
set(WAMR_BUILD_LIB_PTHREAD 0)
set(WAMR_BUILD_WASM_SPEC1 1)
set(WAMR_BUILD_INSTRUCTION_METERING 1)
set(WAMR_BUILD_MEMORY_PROFILING 1)
set(WAMR_BUILD_SHRUNK_MEMORY 1)
set(WAMR_BUILD_LOG 0)
set(WAMR_DISABLE_HW_BOUND_CHECK 1)
set(SHARED_PLATFORM_CONFIG "${CMAKE_CURRENT_SOURCE_DIR}/platform.cmake")
include(${WAMR_ROOT_DIR}/build-scripts/runtime_lib.cmake)
add_library(robot_wamr STATIC ${WAMR_RUNTIME_LIB_SOURCE})
target_compile_options(robot_wamr PRIVATE -mcpu=cortex-m33 -mthumb -mfpu=fpv5-sp-d16 -mfloat-abi=hard -Os -g -ffunction-sections -fdata-sections -fstack-usage)
target_include_directories(robot_wamr PRIVATE
  "${FW_ROOT}/Core/Inc"
  "${FW_ROOT}/Middlewares/Third_Party/FreeRTOS/Source/include"
  "${FW_ROOT}/Middlewares/Third_Party/FreeRTOS/Source/portable/GCC/ARM_CM33_NTZ/non_secure")
